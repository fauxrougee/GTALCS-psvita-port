# PS Vita build of reLCS (VitaSDK).
#   cmake -S vita -B build-vita -G Ninja -DCMAKE_BUILD_TYPE=Release
#   cmake --build build-vita
# Produces reLCS.vpk. Game data goes in ux0:data/reLCS/.
cmake_minimum_required(VERSION 3.20)

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
	if(DEFINED ENV{VITASDK})
		set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
	else()
		message(FATAL_ERROR "Please define VITASDK to point to your SDK path!")
	endif()
endif()

project(reLCS C CXX)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)

set(VITA_APP_NAME "GTA Liberty City Stories")
set(VITA_TITLEID  "RELCS0001")
set(VITA_VERSION  "01.15")
set(VITA_DATA_DIR "ux0:data/reLCS/")
option(VITA_DETAILED_GL_PROFILE "Time every GL draw/state call (adds CPU overhead)" OFF)

# Both executables are built from source in the same dependency graph.
option(VITA_WITH_INTRO "Include the complete movie and isolated launcher" ON)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)

if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
endif()

get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LIBRW "${ROOT}/vendor/librw")

set(VITA_CPU_FLAGS -mcpu=cortex-a9 -mfpu=neon -mtune=cortex-a9)

# ---- librw (OpenGL ES 2.0 via vitaGL, window/context through glfw_vita) ----
# Vita fixes to the librw submodule (PSP2-guarded), applied unless already there.
# --ignore-whitespace: the checkout may have CRLF line endings.
set(LIBRW_PATCH "${CMAKE_CURRENT_SOURCE_DIR}/librw-psp2.patch")
execute_process(COMMAND git apply --directory=vendor/librw --ignore-whitespace --reverse --check "${LIBRW_PATCH}"
	WORKING_DIRECTORY "${ROOT}" RESULT_VARIABLE LIBRW_PATCHED OUTPUT_QUIET ERROR_QUIET)
if(NOT LIBRW_PATCHED EQUAL 0)
	execute_process(COMMAND git apply --directory=vendor/librw --ignore-whitespace "${LIBRW_PATCH}"
		WORKING_DIRECTORY "${ROOT}" RESULT_VARIABLE LIBRW_PATCH_RESULT)
	if(NOT LIBRW_PATCH_RESULT EQUAL 0)
		message(FATAL_ERROR "Could not apply ${LIBRW_PATCH} to ${LIBRW}")
	endif()
	message(STATUS "Applied librw-psp2.patch")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LIBRW_PATCH}")

file(GLOB LIBRW_SOURCES CONFIGURE_DEPENDS
	"${LIBRW}/src/*.cpp"
	"${LIBRW}/src/*/*.cpp"
	"${LIBRW}/src/gl/glad/*.c")
add_library(rw STATIC ${LIBRW_SOURCES})
target_include_directories(rw PUBLIC "${LIBRW}" PRIVATE "${ROOT}/src/skel/vita")
target_compile_definitions(rw PUBLIC RW_GL3 PSP2)
target_compile_options(rw PRIVATE ${VITA_CPU_FLAGS} -fno-strict-aliasing
	$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions -fno-rtti>)

# ---- reLCS ----
set(GAME_DIRS
	src src/animation src/audio src/audio/eax src/audio/oal src/buildings
	src/collision src/control src/core src/entities src/leeds src/leeds/base
	src/math src/modelinfo src/objects src/peds src/renderer src/rw src/save
	src/skel src/skel/glfw src/skel/vita src/text src/vehicles src/weapons
	src/extras src/fakerw)

set(GAME_SOURCES "")
set(GAME_INCLUDES "")
foreach(dir ${GAME_DIRS})
	file(GLOB dir_sources CONFIGURE_DEPENDS "${ROOT}/${dir}/*.cpp" "${ROOT}/${dir}/*.c")
	list(APPEND GAME_SOURCES ${dir_sources})
	list(APPEND GAME_INCLUDES "${ROOT}/${dir}")
endforeach()
# Generated by printHash on desktop builds; versioning is off here.
list(REMOVE_ITEM GAME_SOURCES "${ROOT}/src/extras/GitSHA1.cpp")

add_executable(${PROJECT_NAME} ${GAME_SOURCES})
target_include_directories(${PROJECT_NAME} PRIVATE ${GAME_INCLUDES})
file(SHA256 "$ENV{VITASDK}/arm-vita-eabi/lib/libvitaGL.a" VITA_GL_LIBRARY_HASH)
file(SHA256 "${LIBRW}/src/gl/gl3shader.cpp" VITA_BINDING_HASH)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
	"$ENV{VITASDK}/arm-vita-eabi/lib/libvitaGL.a" "${LIBRW}/src/gl/gl3shader.cpp")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/vita_cache_build.h"
	CONTENT "#define VITA_GL_CACHE_BUILD \"${VITA_GL_LIBRARY_HASH}-${VITA_BINDING_HASH}\"\n")
# Build id in every measurement report (src/skel/vita/vita_perf.cpp): version,
# commit, uncommitted changes and configure time.
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${ROOT}"
	OUTPUT_VARIABLE VITA_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE VITA_GIT_RESULT)
if(NOT VITA_GIT_RESULT EQUAL 0)
	set(VITA_GIT_SHA "nogit")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=no WORKING_DIRECTORY "${ROOT}"
	OUTPUT_VARIABLE VITA_GIT_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(VITA_GIT_DIRTY)
	set(VITA_GIT_SHA "${VITA_GIT_SHA}+dirty")
endif()
string(TIMESTAMP VITA_CONFIGURE_TIME "%Y%m%d-%H%M")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/vita_build_id.h"
	CONTENT "#define VITA_BUILD_ID \"${VITA_VERSION}-${VITA_GIT_SHA}-${VITA_CONFIGURE_TIME}\"\n")
target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(${PROJECT_NAME} PRIVATE LIBRW AUDIO_OAL
	VITA_DATA_DIR="${VITA_DATA_DIR}")
if(VITA_DETAILED_GL_PROFILE)
	target_compile_definitions(${PROJECT_NAME} PRIVATE VITA_DETAILED_GL_PROFILE)
endif()
target_compile_options(${PROJECT_NAME} PRIVATE ${VITA_CPU_FLAGS}
	-fno-strict-aliasing -Wno-write-strings
	$<$<COMPILE_LANGUAGE:CXX>:-fpermissive -std=gnu++17>)

target_link_options(${PROJECT_NAME} PRIVATE
	"-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/sce-metadata-gap.ld"
	# Skip only vitaGL's optional logo thread, keeping the installed library.
	-Wl,--wrap=invoke_splashscreen
	# Console diagnostics go to the bounded async writer (vita_log.cpp).
	-Wl,--wrap=printf,--wrap=vprintf,--wrap=puts,--wrap=putchar
	-Wl,--wrap=fprintf,--wrap=vfprintf,--wrap=fputs,--wrap=fputc,--wrap=fwrite,--wrap=fflush
	# logs vitaGL's out of memory sleeps (src/skel/vita/gl_vita.cpp)
	-Wl,--wrap=sceKernelDelayThread
	# allocation counts in the measurement reports (src/skel/vita/vita_perf.cpp)
	-Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=memalign,--wrap=free)
set_property(TARGET ${PROJECT_NAME} APPEND PROPERTY LINK_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/sce-metadata-gap.ld")

target_link_libraries(${PROJECT_NAME}
	rw
	openal
	mpg123
	vitaGL
	vitashark
	SceShaccCgExt
	mathneon
	taihen_stub
	kubridge_stub
	SceShaccCg_stub
	SceKernelDmacMgr_stub
	SceGxm_stub
	SceDisplay_stub
	SceCommonDialog_stub
	SceIme_stub
	SceAppMgr_stub
	SceAppUtil_stub
	SceSysmodule_stub
	SceCtrl_stub
	SceTouch_stub
	SceMotion_stub
	SceHid_stub
	SceAudio_stub
	SceAudioIn_stub
	ScePower_stub
	SceRtc_stub
	SceLibKernel_stub
	# whole pthread archive, or libstdc++ thinks threads are off and std::thread
	# (MULTITHREADED_AUDIO) aborts with "Enable multithreading to use std::thread"
	-Wl,--whole-archive pthread -Wl,--no-whole-archive
	z
	m
	stdc++
)

vita_create_self(game.bin ${PROJECT_NAME} UNSAFE)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/loading.rgba.z"
	COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/vita/make-loading-art.py"
		--output "${CMAKE_CURRENT_BINARY_DIR}/loading.rgba.z"
	DEPENDS "${ROOT}/tools/vita/make-loading-art.py" sce_sys/livearea/contents/bg.png
	VERBATIM)
set(PACKAGE_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/game.bin"
	"${CMAKE_CURRENT_BINARY_DIR}/loading.rgba.z")
set(PACKAGE_OPTIONS "")
if(VITA_WITH_INTRO)
	include("${CMAKE_CURRENT_SOURCE_DIR}/launcher/build.cmake")
	list(APPEND PACKAGE_OPTIONS --intro)
	list(APPEND PACKAGE_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/intro.bin"
		"${CMAKE_CURRENT_BINARY_DIR}/intro.vtm" "${CMAKE_CURRENT_BINARY_DIR}/intro.json")
endif()
file(GLOB_RECURSE PACKAGE_ART CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/sce_sys/*")
# Keep the working release's extended memory budget. Python packaging also
# handles Windows source/build paths containing spaces without shell splitting.
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/reLCS.vpk"
	COMMAND "${VITA_MKSFOEX}" -d ATTRIBUTE2=12 -s "APP_VER=${VITA_VERSION}"
		-s "TITLE_ID=${VITA_TITLEID}" "${VITA_APP_NAME}" "${CMAKE_CURRENT_BINARY_DIR}/param.sfo"
	COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/vita/package-release.py"
		--build-dir "${CMAKE_CURRENT_BINARY_DIR}" --version "${VITA_VERSION}" ${PACKAGE_OPTIONS}
	DEPENDS ${PACKAGE_DEPENDS} ${PACKAGE_ART} "${ROOT}/tools/vita/package-release.py"
		"${CMAKE_CURRENT_SOURCE_DIR}/launcher/lz4/LICENSE"
	VERBATIM)
add_custom_target(vpk ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/reLCS.vpk")
